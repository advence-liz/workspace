# 09 - 错误处理与限流

> 生产环境必备：优雅处理错误和速率限制

## 🚨 常见错误类型

### 1. API 错误码

| 状态码 | 错误类型 | 说明 | 处理方式 |
|--------|---------|------|---------|
| **400** | Bad Request | 请求格式错误 | 检查参数 |
| **401** | Unauthorized | API Key 无效 | 验证密钥 |
| **403** | Forbidden | 无权访问 | 检查权限 |
| **404** | Not Found | 资源不存在 | 检查端点 |
| **429** | Rate Limit | 速率限制 | 重试 |
| **500** | Server Error | 服务器错误 | 重试 |
| **503** | Service Unavailable | 服务不可用 | 等待重试 |

## 🛡️ 错误处理策略

### 1. 基础错误处理

```typescript
import { OpenAI } from 'openai';

async function safeChat(prompt: string) {
  try {
    const response = await openai.chat.completions.create({
      model: 'gpt-3.5-turbo',
      messages: [{ role: 'user', content: prompt }],
    });

    return response.choices[0].message.content;
  } catch (error: any) {
    // 检查是否是 OpenAI API 错误
    if (error instanceof OpenAI.APIError) {
      console.error('OpenAI API 错误:');
      console.error('  状态码:', error.status);
      console.error('  消息:', error.message);
      console.error('  类型:', error.type);
      console.error('  代码:', error.code);

      // 根据错误类型处理
      switch (error.status) {
        case 400:
          throw new Error('请求参数错误，请检查输入');
        case 401:
          throw new Error('API Key 无效，请检查配置');
        case 429:
          throw new Error('请求过于频繁，请稍后重试');
        case 500:
        case 503:
          throw new Error('OpenAI 服务暂时不可用');
        default:
          throw new Error(`API 错误: ${error.message}`);
      }
    }

    // 网络错误
    if (error.code === 'ECONNREFUSED' || error.code === 'ENOTFOUND') {
      throw new Error('网络连接失败，请检查网络');
    }

    // 超时错误
    if (error.code === 'ETIMEDOUT') {
      throw new Error('请求超时，请重试');
    }

    // 未知错误
    throw new Error(`未知错误: ${error.message}`);
  }
}
```

### 2. 分类错误处理器

```typescript
class OpenAIErrorHandler {
  static handle(error: any): never {
    if (error instanceof OpenAI.APIError) {
      return this.handleAPIError(error);
    }

    if (error.code) {
      return this.handleNetworkError(error);
    }

    throw new Error(`未处理的错误: ${error.message}`);
  }

  private static handleAPIError(error: OpenAI.APIError): never {
    const errorMap: Record<number, string> = {
      400: '请求格式错误',
      401: 'API Key 无效',
      403: '无访问权限',
      404: '资源不存在',
      429: '请求过于频繁',
      500: '服务器内部错误',
      503: '服务暂时不可用',
    };

    const message = errorMap[error.status] || `API 错误 (${error.status})`;
    throw new Error(`${message}: ${error.message}`);
  }

  private static handleNetworkError(error: any): never {
    const networkErrors: Record<string, string> = {
      ECONNREFUSED: '连接被拒绝',
      ENOTFOUND: '无法解析域名',
      ETIMEDOUT: '请求超时',
      ECONNRESET: '连接被重置',
    };

    const message = networkErrors[error.code] || '网络错误';
    throw new Error(`${message}: ${error.message}`);
  }
}

// 使用
try {
  const result = await openai.chat.completions.create({...});
} catch (error) {
  OpenAIErrorHandler.handle(error);
}
```

## 🔄 重试机制

### 1. 简单重试

```typescript
async function retryRequest<T>(
  fn: () => Promise<T>,
  maxRetries: number = 3
): Promise<T> {
  let lastError: any;

  for (let i = 0; i < maxRetries; i++) {
    try {
      return await fn();
    } catch (error: any) {
      lastError = error;

      // 只重试可重试的错误
      const retryableStatuses = [429, 500, 502, 503, 504];
      const isRetryable =
        error instanceof OpenAI.APIError &&
        retryableStatuses.includes(error.status);

      if (!isRetryable || i === maxRetries - 1) {
        throw error;
      }

      console.log(`⏳ 第 ${i + 1} 次重试...`);
      await new Promise((resolve) => setTimeout(resolve, 1000));
    }
  }

  throw lastError;
}

// 使用
const result = await retryRequest(() =>
  openai.chat.completions.create({
    model: 'gpt-3.5-turbo',
    messages: [{ role: 'user', content: 'Hello' }],
  })
);
```

### 2. 指数退避重试

```typescript
async function retryWithExponentialBackoff<T>(
  fn: () => Promise<T>,
  options: {
    maxRetries?: number;
    baseDelay?: number;
    maxDelay?: number;
  } = {}
): Promise<T> {
  const { maxRetries = 5, baseDelay = 1000, maxDelay = 32000 } = options;

  for (let i = 0; i < maxRetries; i++) {
    try {
      return await fn();
    } catch (error: any) {
      // 最后一次尝试
      if (i === maxRetries - 1) throw error;

      // 检查是否可重试
      const retryableStatuses = [429, 500, 502, 503, 504];
      const isRetryable =
        error instanceof OpenAI.APIError &&
        retryableStatuses.includes(error.status);

      if (!isRetryable) throw error;

      // 计算延迟时间（指数退避 + 随机抖动）
      const exponentialDelay = Math.min(baseDelay * Math.pow(2, i), maxDelay);
      const jitter = Math.random() * 1000; // 0-1000ms 随机抖动
      const delay = exponentialDelay + jitter;

      console.log(`⏳ 等待 ${Math.round(delay)}ms 后第 ${i + 1} 次重试...`);
      await new Promise((resolve) => setTimeout(resolve, delay));
    }
  }

  throw new Error('不应该到达这里');
}
```

### 3. 带重试头的智能重试

```typescript
async function retryWithHeaders<T>(
  fn: () => Promise<T>,
  maxRetries: number = 3
): Promise<T> {
  for (let i = 0; i < maxRetries; i++) {
    try {
      return await fn();
    } catch (error: any) {
      if (i === maxRetries - 1) throw error;

      // 检查 429 错误
      if (error instanceof OpenAI.APIError && error.status === 429) {
        // 尝试从响应头获取重试时间
        const retryAfter = error.headers?.['retry-after'];

        if (retryAfter) {
          const delay = parseInt(retryAfter) * 1000;
          console.log(`⏳ 服务器建议等待 ${delay}ms`);
          await new Promise((resolve) => setTimeout(resolve, delay));
        } else {
          // 使用默认延迟
          const delay = 1000 * Math.pow(2, i);
          await new Promise((resolve) => setTimeout(resolve, delay));
        }
      } else if (error.status >= 500) {
        // 服务器错误，使用指数退避
        const delay = 1000 * Math.pow(2, i);
        await new Promise((resolve) => setTimeout(resolve, delay));
      } else {
        // 不可重试的错误
        throw error;
      }
    }
  }

  throw new Error('重试次数耗尽');
}
```

## ⏱️ 速率限制管理

### 1. 简单的速率限制器

```typescript
class RateLimiter {
  private queue: Array<() => Promise<any>> = [];
  private running = 0;

  constructor(
    private maxConcurrent: number = 5,
    private minInterval: number = 200 // 最小请求间隔（毫秒）
  ) {}

  async execute<T>(fn: () => Promise<T>): Promise<T> {
    // 等待空闲槽位
    while (this.running >= this.maxConcurrent) {
      await new Promise((resolve) => setTimeout(resolve, 100));
    }

    this.running++;

    try {
      const result = await fn();

      // 等待最小间隔
      await new Promise((resolve) => setTimeout(resolve, this.minInterval));

      return result;
    } finally {
      this.running--;
    }
  }
}

// 使用
const limiter = new RateLimiter(3, 500); // 最多 3 个并发，间隔 500ms

const results = await Promise.all(
  prompts.map((prompt) =>
    limiter.execute(() =>
      openai.chat.completions.create({
        model: 'gpt-3.5-turbo',
        messages: [{ role: 'user', content: prompt }],
      })
    )
  )
);
```

### 2. Token Bucket 算法

```typescript
class TokenBucket {
  private tokens: number;
  private lastRefill: number;

  constructor(
    private capacity: number, // 桶容量
    private refillRate: number // 每秒补充的 token 数
  ) {
    this.tokens = capacity;
    this.lastRefill = Date.now();
  }

  private refill() {
    const now = Date.now();
    const timePassed = (now - this.lastRefill) / 1000; // 秒
    const tokensToAdd = timePassed * this.refillRate;

    this.tokens = Math.min(this.capacity, this.tokens + tokensToAdd);
    this.lastRefill = now;
  }

  async consume(tokens: number = 1): Promise<void> {
    this.refill();

    if (this.tokens >= tokens) {
      this.tokens -= tokens;
      return;
    }

    // 计算需要等待的时间
    const tokensNeeded = tokens - this.tokens;
    const waitTime = (tokensNeeded / this.refillRate) * 1000;

    console.log(`⏳ 速率限制，等待 ${Math.round(waitTime)}ms`);
    await new Promise((resolve) => setTimeout(resolve, waitTime));

    // 重新尝试
    return this.consume(tokens);
  }
}

// 使用（假设限制为每分钟 60 个请求）
const bucket = new TokenBucket(60, 1); // 容量 60，每秒补充 1 个

async function rateLimitedChat(prompt: string) {
  await bucket.consume(1);

  return openai.chat.completions.create({
    model: 'gpt-3.5-turbo',
    messages: [{ role: 'user', content: prompt }],
  });
}
```

### 3. 队列系统

```typescript
class RequestQueue {
  private queue: Array<{
    fn: () => Promise<any>;
    resolve: (value: any) => void;
    reject: (error: any) => void;
  }> = [];
  private processing = false;

  constructor(private delayBetweenRequests: number = 1000) {}

  async enqueue<T>(fn: () => Promise<T>): Promise<T> {
    return new Promise((resolve, reject) => {
      this.queue.push({ fn, resolve, reject });
      this.process();
    });
  }

  private async process() {
    if (this.processing || this.queue.length === 0) return;

    this.processing = true;

    while (this.queue.length > 0) {
      const { fn, resolve, reject } = this.queue.shift()!;

      try {
        const result = await fn();
        resolve(result);
      } catch (error) {
        reject(error);
      }

      // 等待间隔
      if (this.queue.length > 0) {
        await new Promise((resolve) =>
          setTimeout(resolve, this.delayBetweenRequests)
        );
      }
    }

    this.processing = false;
  }

  getQueueLength(): number {
    return this.queue.length;
  }
}

// 使用
const queue = new RequestQueue(500);

const result = await queue.enqueue(() =>
  openai.chat.completions.create({
    model: 'gpt-3.5-turbo',
    messages: [{ role: 'user', content: 'Hello' }],
  })
);
```

## 🔧 完整的生产级封装

```typescript
import { OpenAI } from 'openai';

interface RequestOptions {
  maxRetries?: number;
  timeout?: number;
  rateLimitPerMinute?: number;
}

class ProductionOpenAI {
  private openai: OpenAI;
  private limiter: RateLimiter;

  constructor(apiKey: string, options: RequestOptions = {}) {
    this.openai = new OpenAI({ apiKey });
    this.limiter = new RateLimiter(
      5,
      60000 / (options.rateLimitPerMinute || 60)
    );
  }

  async chat(
    messages: OpenAI.Chat.ChatCompletionMessageParam[],
    options: Partial<OpenAI.Chat.ChatCompletionCreateParams> = {}
  ): Promise<string> {
    return this.limiter.execute(() =>
      retryWithExponentialBackoff(
        async () => {
          try {
            const response = await this.openai.chat.completions.create({
              model: 'gpt-3.5-turbo',
              messages,
              ...options,
            });

            return response.choices[0].message.content || '';
          } catch (error) {
            // 记录错误
            console.error('Chat completion 错误:', error);
            throw error;
          }
        },
        { maxRetries: 3 }
      )
    );
  }

  async embedding(text: string): Promise<number[]> {
    return this.limiter.execute(() =>
      retryWithExponentialBackoff(async () => {
        const response = await this.openai.embeddings.create({
          model: 'text-embedding-3-small',
          input: text,
        });

        return response.data[0].embedding;
      })
    );
  }
}

// 使用
const ai = new ProductionOpenAI(process.env.OPENAI_API_KEY!, {
  maxRetries: 3,
  rateLimitPerMinute: 60,
});

const response = await ai.chat([
  { role: 'user', content: 'Hello!' }
]);
```

## 📊 监控与告警

```typescript
interface ErrorStats {
  total: number;
  byStatus: Record<number, number>;
  byType: Record<string, number>;
}

class ErrorMonitor {
  private stats: ErrorStats = {
    total: 0,
    byStatus: {},
    byType: {},
  };

  record(error: any) {
    this.stats.total++;

    if (error instanceof OpenAI.APIError) {
      this.stats.byStatus[error.status] =
        (this.stats.byStatus[error.status] || 0) + 1;
      this.stats.byType[error.type] =
        (this.stats.byType[error.type] || 0) + 1;
    }

    // 检查是否需要告警
    if (this.shouldAlert()) {
      this.sendAlert();
    }
  }

  private shouldAlert(): boolean {
    // 5 分钟内超过 10 个错误
    return this.stats.total > 10;
  }

  private sendAlert() {
    console.error('🚨 错误率过高！', this.stats);
    // 发送告警到监控系统
  }

  getStats(): ErrorStats {
    return { ...this.stats };
  }

  reset() {
    this.stats = { total: 0, byStatus: {}, byType: {} };
  }
}

// 使用
const monitor = new ErrorMonitor();

try {
  await openai.chat.completions.create({...});
} catch (error) {
  monitor.record(error);
  throw error;
}
```
