### 2. 输入验证

```typescript
import { z } from 'zod';

const UserInputSchema = z.object({
  prompt: z.string().min(1).max(1000),
  temperature: z.number().min(0).max(2).optional(),
});

async function safeChat(userInput: unknown) {
  // 验证输入
  const validated = UserInputSchema.parse(userInput);

  // 清理输入（防止注入攻击）
  const sanitizedPrompt = validated.prompt
    .replace(/<script>/gi, '')
    .trim();

  return await openai.chat.completions.create({
    model: 'gpt-3.5-turbo',
    messages: [{ role: 'user', content: sanitizedPrompt }],
    temperature: validated.temperature ?? 0.7,
  });
}
```

### 3. 缓存策略

```typescript
import NodeCache from 'node-cache';

const cache = new NodeCache({ stdTTL: 3600 }); // 1 小时过期

async function cachedCompletion(
  prompt: string,
  options: OpenAI.Chat.ChatCompletionCreateParams
) {
  const cacheKey = `${prompt}-${JSON.stringify(options)}`;

  // 检查缓存
  const cached = cache.get<string>(cacheKey);
  if (cached) {
    console.log('💾 缓存命中');
    return cached;
  }

  // 调用 API
  const response = await openai.chat.completions.create({
    ...options,
    messages: [{ role: 'user', content: prompt }],
  });

  const result = response.choices[0].message.content!;

  // 存入缓存
  cache.set(cacheKey, result);

  return result;
}
```

### 4. 批量处理

```typescript
async function batchProcess(prompts: string[]) {
  // 并发控制
  const batchSize = 5;
  const results: string[] = [];

  for (let i = 0; i < prompts.length; i += batchSize) {
    const batch = prompts.slice(i, i + batchSize);

    const responses = await Promise.all(
      batch.map(prompt =>
        openai.chat.completions.create({
          model: 'gpt-3.5-turbo',
          messages: [{ role: 'user', content: prompt }],
        })
      )
    );

    results.push(...responses.map(r => r.choices[0].message.content!));

    // 避免速率限制
    if (i + batchSize < prompts.length) {
      await new Promise(resolve => setTimeout(resolve, 1000));
    }
  }

  return results;
}
```

async function streamToClient(prompt: string, res: Response) {
  const stream = await openai.chat.completions.create({
    model: 'gpt-3.5-turbo',
    messages: [{ role: 'user', content: prompt }],
    stream: true,
  });

  res.setHeader('Content-Type', 'text/event-stream');
  res.setHeader('Cache-Control', 'no-cache');
  res.setHeader('Connection', 'keep-alive');

  for await (const chunk of stream) {

  }

  res.write('data: [DONE]\n\n');
  res.end();
}

## 🔄 错误处理

### 1. 完整的错误处理

```typescript
import { OpenAI } from 'openai';

async function robustChat(prompt: string) {
  try {
    const response = await openai.chat.completions.create({
      model: 'gpt-3.5-turbo',
      messages: [{ role: 'user', content: prompt }],
    });

    return response.choices[0].message.content;

  } catch (error: any) {
    // API 错误
    if (error instanceof OpenAI.APIError) {
      console.error('API 错误:', error.status, error.message);

      // 速率限制
      if (error.status === 429) {
        console.log('⏳ 速率限制，等待后重试...');
        await new Promise(resolve => setTimeout(resolve, 5000));
        return robustChat(prompt); // 重试
      }

      // 无效请求
      if (error.status === 400) {
        throw new Error(`请求无效: ${error.message}`);
      }

      // 认证失败
      if (error.status === 401) {
        throw new Error('API Key 无效');
      }

      // 服务器错误
      if (error.status >= 500) {
        throw new Error('OpenAI 服务暂时不可用');
      }
    }

    // 网络错误
    if (error.code === 'ECONNREFUSED') {
      throw new Error('网络连接失败');
    }

    // 其他错误
    throw error;
  }
}
```

### 2. 重试机制

```typescript
async function retryWithBackoff<T>(
  fn: () => Promise<T>,
  maxRetries: number = 3,
  baseDelay: number = 1000
): Promise<T> {
  for (let i = 0; i < maxRetries; i++) {
    try {
      return await fn();
    } catch (error: any) {
      // 最后一次尝试失败，抛出错误
      if (i === maxRetries - 1) throw error;

      // 只重试临时性错误
      if (error.status === 429 || error.status >= 500) {
        const delay = baseDelay * Math.pow(2, i); // 指数退避
        console.log(`⏳ 第 ${i + 1} 次重试，等待 ${delay}ms`);
        await new Promise(resolve => setTimeout(resolve, delay));
      } else {
        throw error; // 不可重试的错误
      }
    }
  }

  throw new Error('不应该到达这里');
}

// 使用
const result = await retryWithBackoff(() =>
  openai.chat.completions.create({
    model: 'gpt-3.5-turbo',
    messages: [{ role: 'user', content: 'Hello' }],
  })
);
```

### 2. 性能监控

```typescript
class OpenAIMonitor {
  private totalRequests = 0;
  private totalTokens = 0;
  private totalCost = 0;
  private errors = 0;

  async track<T>(fn: () => Promise<T>): Promise<T> {
    this.totalRequests++;

    try {
      const result = await fn();

      // 提取使用信息（如果是 completion 响应）
      if (result && typeof result === 'object' && 'usage' in result) {
        const usage = (result as any).usage;
        this.totalTokens += usage.total_tokens;
        this.totalCost += this.calculateCost(usage);
      }

      return result;
    } catch (error) {
      this.errors++;
      throw error;
    }
  }

  private calculateCost(usage: any): number {
    // 简化的成本计算
    return (usage.total_tokens * 0.002) / 1000;
  }

  getStats() {
    return {
      totalRequests: this.totalRequests,
      totalTokens: this.totalTokens,
      totalCost: this.totalCost.toFixed(4),
      errors: this.errors,
      successRate: ((this.totalRequests - this.errors) / this.totalRequests * 100).toFixed(2) + '%',
    };
  }
}

// 使用
const monitor = new OpenAIMonitor();

const response = await monitor.track(() =>
  openai.chat.completions.create({
    model: 'gpt-3.5-turbo',
    messages: [{ role: 'user', content: 'Hello' }],
  })
);

console.log('📈 统计:', monitor.getStats());
```
